cmake_minimum_required(VERSION 3.0.2)
project(sensor_array_bringup)
find_package(catkin REQUIRED COMPONENTS message_generation std_msgs maggrad_protocol)
catkin_python_setup()
add_message_files(FILES SensorCorrectionSample.msg SensorCorrectionRound.msg)
generate_messages(DEPENDENCIES std_msgs maggrad_protocol)
catkin_package(CATKIN_DEPENDS message_runtime std_msgs maggrad_protocol)
catkin_install_python(PROGRAMS scripts/bringup scripts/configured_bringup scripts/sensor_correction_node
  DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES README.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_nosetests(test)
endif()
